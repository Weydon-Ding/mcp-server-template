> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a reusable TypeScript/Node.js MCP server template with shared business logic exposed through stdio MCP, Streamable HTTP MCP, and a direct CLI.

**Architecture:** Implement domain tools as pure functions with Zod schemas, then expose them through thin MCP and CLI adapters. Use one MCP server factory shared by both transports; stdio and HTTP only handle protocol startup, configuration, and auth. Keep tests layered: domain unit tests, CLI behavior tests, and MCP smoke tests.

**Tech Stack:** Node.js, TypeScript, pnpm, `@modelcontextprotocol/sdk`, commander, Zod, Vitest, ESLint, Prettier, Express.

export const ErrorCode = {
  Validation: "VALIDATION_ERROR",
  Config: "CONFIG_ERROR",
  Runtime: "RUNTIME_ERROR",
  Unknown: "UNKNOWN_ERROR"
} as const;

export type ErrorCode = (typeof ErrorCode)[keyof typeof ErrorCode];

export class DomainError extends Error {
  readonly code: ErrorCode;
  readonly cause?: unknown;

  constructor(code: ErrorCode, message: string, cause?: unknown) {
    super(message);
    this.name = "DomainError";
    this.code = code;
    this.cause = cause;
  }
}

export function getExitCodeForError(code: ErrorCode): 1 | 2 {
  switch (code) {
    case ErrorCode.Validation:
    case ErrorCode.Config:
      return 2;
    case ErrorCode.Runtime:
    case ErrorCode.Unknown:
      return 1;
  }
}

import { ZodError } from "zod";
import { DomainError, ErrorCode, getExitCodeForError } from "../../domain/errors.js";

export type CliError = {
  code: ErrorCode;
  message: string;
  exitCode: 1 | 2;
};

export function toCliError(error: unknown): CliError {
  if (error instanceof DomainError) {
    return {
      code: error.code,
      message: error.message,
      exitCode: getExitCodeForError(error.code)
    };
  }

  if (error instanceof ZodError) {
    return {
      code: ErrorCode.Validation,
      message: error.issues.map((issue) => issue.message).join("; "),
      exitCode: 2
    };
  }

  if (error instanceof Error) {
    return {
      code: ErrorCode.Unknown,
      message: error.message,
      exitCode: 1
    };
  }

  return {
    code: ErrorCode.Unknown,
    message: "Unknown error",
    exitCode: 1
  };
}

Create `src/adapters/cli/commands/tools.ts`:

```ts
import { Command } from "commander";
import { ErrorCode } from "../../../domain/errors.js";
import { runAdd } from "../../../domain/tools/add.js";
import { runEcho } from "../../../domain/tools/echo.js";
import { toCliError } from "../errors.js";
import { formatJsonError, formatJsonSuccess } from "../format.js";

export type Output = {
  writeStdout(text: string): void;
  writeStderr(text: string): void;
  exit(code: number): never;
};

function parseCliNumber(value: string, name: string): number {
  const parsed = Number(value);
  if (!Number.isFinite(parsed)) {
    throw new Error(`${name} must be a finite number`);
  }
  return parsed;
}

function handleError(error: unknown, json: boolean, output: Output): never {
  const cliError = error instanceof Error && error.message.includes("must be a finite number")
    ? { code: ErrorCode.Validation, message: error.message, exitCode: 2 as const }
    : toCliError(error);

  if (json) {
    output.writeStdout(formatJsonError(cliError.code, cliError.message));
  } else {
    output.writeStderr(`${cliError.code}: ${cliError.message}`);
  }

  return output.exit(cliError.exitCode);
}

export function createToolsCommand(output: Output): Command {
  const tools = new Command("tools").description("Run business tools directly from the CLI");

  tools
    .command("echo")
    .description("Return the provided text unchanged")
    .argument("<text>", "Text to echo")
    .option("--json", "Print structured JSON output")
    .action(async (text: string, options: { json?: boolean }) => {
      const json = options.json === true;
      try {
        const result = await runEcho({ text });
        output.writeStdout(json ? formatJsonSuccess(result) : result.text);
      } catch (error) {
        handleError(error, json, output);
      }
    });

  tools
    .command("add")
    .description("Add two numbers")
    .argument("<a>", "First number")
    .argument("<b>", "Second number")
    .option("--json", "Print structured JSON output")
    .action(async (a: string, b: string, options: { json?: boolean }) => {
      const json = options.json === true;
      try {
        const result = await runAdd({ a: parseCliNumber(a, "a"), b: parseCliNumber(b, "b") });
        output.writeStdout(json ? formatJsonSuccess(result) : String(result.result));
      } catch (error) {
        handleError(error, json, output);
      }
    });

  return tools;
}
```

export type HttpConfig = {
  host: string;
  port: number;
  token?: string;
};

export type HttpConfigOptions = {
  host?: string;
  port?: string;
};

function parsePort(value: string): number {
  const port = Number(value);
  if (!Number.isInteger(port) || port < 0 || port > 65_535) {
    throw new DomainError(ErrorCode.Config, `Invalid MCP HTTP port: ${value}`);
  }
  return port;
}

export function parseHttpConfig(
  options: HttpConfigOptions = {},
  env: NodeJS.ProcessEnv = process.env,
): HttpConfig {
  const host = options.host ?? env.MCP_HTTP_HOST ?? "127.0.0.1";
  const port = parsePort(options.port ?? env.MCP_HTTP_PORT ?? "3000");
  const token = env.MCP_HTTP_TOKEN && env.MCP_HTTP_TOKEN.length > 0 ? env.MCP_HTTP_TOKEN : undefined;

  return { host, port, token };
}

import { StreamableHTTPServerTransport } from "@modelcontextprotocol/sdk/server/streamableHttp.js";
import express, { type Express, type Request, type Response } from "express";
import { createMcpServer } from "../adapters/mcp/create-server.js";
import { parseHttpConfig, type HttpConfig, type HttpConfigOptions } from "../config/http.js";

export type HttpCliOptions = HttpConfigOptions;

function isAuthorized(req: Request, token: string | undefined): boolean {
  if (!token) {
    return true;
  }
  return req.header("authorization") === `Bearer ${token}`;
}

export function createHttpApp(config: HttpConfig): Express {
  const app = express();
  app.use(express.json({ limit: "1mb" }));

  app.post("/mcp", async (req: Request, res: Response) => {
    if (!isAuthorized(req, config.token)) {
      res.status(401).json({ error: "unauthorized" });
      return;
    }

    const server = createMcpServer();
    const transport = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined });
    await server.connect(transport);
    await transport.handleRequest(req, res, req.body);
  });

  return app;
}

export async function startHttpTransport(options: HttpCliOptions = {}): Promise<void> {
  const config = parseHttpConfig(options);

  if (config.host === "0.0.0.0" && !config.token) {
    process.stderr.write(
      "WARNING: MCP HTTP server is binding to 0.0.0.0 without MCP_HTTP_TOKEN. Use only on trusted networks.\n",
    );
  }

  const app = createHttpApp(config);
  await new Promise<void>((resolve) => {
    app.listen(config.port, config.host, () => {
      process.stderr.write(`MCP HTTP server listening on http://${config.host}:${config.port}/mcp\n`);
      resolve();
    });
  });
}
